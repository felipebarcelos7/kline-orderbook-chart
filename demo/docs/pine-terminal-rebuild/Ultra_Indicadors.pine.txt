// 05. Liquidation Heatmap proxy
// -----------------------------------------------------------------------------
var box[] liqBoxes = array.new_box()
var bool[] liqLongSides = array.new_bool()
var int[] liqBornBars = array.new_int()
var int liqLongHits = 0
var int liqShortHits = 0

f_liq_add(float anchorPrice, float leverage, bool longSide, int startBar) =>
    float distance = anchorPrice / leverage * 0.92
    float levelPrice = longSide ? anchorPrice - distance : anchorPrice + distance
    float halfHeight = atr14 * 0.10
    box zoneBox = box.new(startBar, levelPrice + halfHeight, bar_index + liqProjection, levelPrice - halfHeight,
         xloc=xloc.bar_index, bgcolor=color.new(longSide ? color.red : color.aqua, leverage == 50 ? 62 : 79),
         border_color=color.new(longSide ? color.red : color.aqua, 58), force_overlay=true)
    array.push(liqBoxes, zoneBox)
    array.push(liqLongSides, longSide)
    array.push(liqBornBars, bar_index)

float liqPivotHigh = ta.pivothigh(high, liqPivotLength, liqPivotLength)
float liqPivotLow = ta.pivotlow(low, liqPivotLength, liqPivotLength)
if useLiq and barstate.isconfirmed
    if not na(liqPivotLow)
        f_liq_add(liqPivotLow, 25, true, bar_index - liqPivotLength)
        f_liq_add(liqPivotLow, 50, true, bar_index - liqPivotLength)
    if not na(liqPivotHigh)
        f_liq_add(liqPivotHigh, 25, false, bar_index - liqPivotLength)
        f_liq_add(liqPivotHigh, 50, false, bar_index - liqPivotLength)
while array.size(liqBoxes) > liqMaxZones
    box.delete(array.shift(liqBoxes))
    array.shift(liqLongSides)
    array.shift(liqBornBars)
bool liqLongHit = false
bool liqShortHit = false
int liqIndex = array.size(liqBoxes) - 1
while liqIndex >= 0
    box zoneBox = array.get(liqBoxes, liqIndex)
    bool longSide = array.get(liqLongSides, liqIndex)
    int bornBar = array.get(liqBornBars, liqIndex)
    bool touched = high >= box.get_bottom(zoneBox) and low <= box.get_top(zoneBox)
    bool expiredZone = bar_index - bornBar > liqProjection
    if touched or expiredZone or not useLiq
        if touched
            if longSide
                liqLongHits += 1
                liqLongHit := true
            else
                liqShortHits += 1
                liqShortHit := true
        box.delete(zoneBox)
        array.remove(liqBoxes, liqIndex)
        array.remove(liqLongSides, liqIndex)
        array.remove(liqBornBars, liqIndex)
    else
        box.set_right(zoneBox, bar_index + 30)
    liqIndex -= 1
plot(useLiq ? close * (1 - 0.92 / 25) : na, "Predicted long liquidation", color=color.new(color.red, 55),
     style=plot.style_linebr, force_overlay=true)
plot(useLiq ? close * (1 + 0.92 / 25) : na, "Predicted short liquidation", color=color.new(color.aqua, 55),
     style=plot.style_linebr, force_overlay=true)

// -----------------------------------------------------------------------------
// 06. VRVP
// -----------------------------------------------------------------------------
var box[] vrvpBoxes = array.new_box()
var line vrvpPocLine = na
var line vrvpVahLine = na
var line vrvpValLine = na
var label vrvpPocLabel = na
var float vrvpPoc = na
var float vrvpVah = na
var float vrvpVal = na
if barstate.islast
    while array.size(vrvpBoxes) > 0
        box.delete(array.pop(vrvpBoxes))
    if not na(vrvpPocLine)
        line.delete(vrvpPocLine)
        vrvpPocLine := na
    if not na(vrvpVahLine)
        line.delete(vrvpVahLine)
        vrvpVahLine := na
    if not na(vrvpValLine)
        line.delete(vrvpValLine)
        vrvpValLine := na
    if not na(vrvpPocLabel)
        label.delete(vrvpPocLabel)
        vrvpPocLabel := na
    if useVrvp
        int vrvpBars = math.min(profileLookback, bar_index + 1)
        float vrvpHigh = high
        float vrvpLow = low
        int vrvpRangeBar = 0
        while vrvpRangeBar < vrvpBars
            vrvpHigh := math.max(vrvpHigh, high[vrvpRangeBar])
            vrvpLow := math.min(vrvpLow, low[vrvpRangeBar])
            vrvpRangeBar += 1
        float vrvpStep = math.max((vrvpHigh - vrvpLow) / profileRows, syminfo.mintick)
        float[] vrvpBuyBins = array.new_float(profileRows, 0)
        float[] vrvpSellBins = array.new_float(profileRows, 0)
        float[] vrvpTotalBins = array.new_float(profileRows, 0)
        float vrvpTotalVolume = 0
        int vrvpBar = 0
        while vrvpBar < vrvpBars
            int binIndex = int(math.floor((hlc3[vrvpBar] - vrvpLow) / vrvpStep))
            binIndex := math.max(0, math.min(profileRows - 1, binIndex))
            float buyRatio = (close[vrvpBar] - low[vrvpBar]) / math.max(high[vrvpBar] - low[vrvpBar], syminfo.mintick)
            array.set(vrvpBuyBins, binIndex, array.get(vrvpBuyBins, binIndex) + volume[vrvpBar] * buyRatio)
            array.set(vrvpSellBins, binIndex, array.get(vrvpSellBins, binIndex) + volume[vrvpBar] * (1 - buyRatio))
            array.set(vrvpTotalBins, binIndex, array.get(vrvpTotalBins, binIndex) + volume[vrvpBar])
            vrvpTotalVolume += volume[vrvpBar]
            vrvpBar += 1
        int vrvpPocIndex = 0
        float vrvpMaxVolume = 0
        int vrvpBin = 0
        while vrvpBin < profileRows
            if array.get(vrvpTotalBins, vrvpBin) > vrvpMaxVolume
                vrvpMaxVolume := array.get(vrvpTotalBins, vrvpBin)
                vrvpPocIndex := vrvpBin
            vrvpBin += 1
        int vrvpVaLowIndex = vrvpPocIndex
        int vrvpVaHighIndex = vrvpPocIndex
        float vrvpVaVolume = array.get(vrvpTotalBins, vrvpPocIndex)
        int vrvpGuard = 0
        while vrvpVaVolume < vrvpTotalVolume * profileValueArea and vrvpGuard < profileRows
            float downVolume = vrvpVaLowIndex > 0 ? array.get(vrvpTotalBins, vrvpVaLowIndex - 1) : -1
            float upVolume = vrvpVaHighIndex < profileRows - 1 ? array.get(vrvpTotalBins, vrvpVaHighIndex + 1) : -1
            if upVolume >= downVolume and upVolume >= 0
